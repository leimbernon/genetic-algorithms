#[test]
#[cfg(feature = "serde")]
fn test_hof_serde_roundtrip() {
    use crate::structures::{Chromosome, Gene};
    use genetic_algorithms::hall_of_fame::*;
    use genetic_algorithms::traits::ChromosomeT;

    // Create test chromosomes with default fitness_fn (not set)
    // so that after serde round-trip, the default matches.
    let c1 = Chromosome {
        dna: vec![Gene { id: 1 }, Gene { id: 2 }],
        fitness: 10.0,
        age: 0,
        fitness_fn: Default::default(),
    };
    let c2 = Chromosome {
        dna: vec![Gene { id: 3 }, Gene { id: 4 }],
        fitness: 5.0,
        age: 0,
        fitness_fn: Default::default(),
    };

    let config = HallOfFameConfig {
        capacity: 10,
        distance_metric: DistanceMetric::Fitness { min_distance: 0.0 },
    };
    let mut hof = HallOfFame::new(config);

    assert!(hof.try_insert(&c1, 0));
    assert!(hof.try_insert(&c2, 5));
    assert_eq!(hof.len(), 2);

    // Serialize
    let json = serde_json::to_string(&hof)
        .expect("Serialization should succeed");

    // Basic sanity: JSON contains expected values
    assert!(json.contains("\"fitness_at_addition\":10.0"));
    assert!(json.contains("\"generation_added\":0"));
    assert!(json.contains("\"capacity\":10"));

    // Deserialize
    let deserialized: HallOfFame<Chromosome> = serde_json::from_str(&json)
        .expect("Deserialization should succeed");

    // Compare structural properties
    assert_eq!(hof.len(), deserialized.len());
    assert_eq!(hof.capacity(), deserialized.capacity());

    // Compare entries chromosome-by-chromosome (dna + fitness + age)
    for (original, deser) in hof.entries().iter().zip(deserialized.entries().iter()) {
        assert_eq!(original.chromosome.dna(), deser.chromosome.dna());
        assert_eq!(original.chromosome.fitness(), deser.chromosome.fitness());
        // age comparison is valid since ChromosomeT::age() returns original age
        assert_eq!(original.generation_added, deser.generation_added);
        assert_eq!(original.fitness_at_addition, deser.fitness_at_addition);
    }

    // Serialize-deserialize-serialize should be idempotent (ignoring fitness_fn)
    let json2 = serde_json::to_string(&deserialized)
        .expect("Second serialization should succeed");
    // The JSON strings may not match exactly if fitness_fn differs,
    // but both should parse to structurally identical HallOfFame entries
    let re_parsed: HallOfFame<Chromosome> = serde_json::from_str(&json2)
        .expect("Second deserialization should succeed");
    assert_eq!(re_parsed.len(), deserialized.len());
}
```

//! Hall of Fame / Solution Archive demo.
//!
//! Demonstrates how to configure and use the Hall of Fame to maintain
//! a bounded archive of the best unique solutions found during a GA run.
//!
//! Run with: cargo run --example hall_of_fame_demo

use genetic_algorithms::chromosomes::Range as RangeChromosome;
use genetic_algorithms::configuration::ProblemSolving;
use genetic_algorithms::ga::Ga;
use genetic_algorithms::genotypes::Range as RangeGene;
use genetic_algorithms::hall_of_fame::{DistanceMetric, HallOfFameConfig};
use genetic_algorithms::initializers::range_random_initialization;
use genetic_algorithms::operations::{Crossover, Mutation, Selection, Survivor};
use genetic_algorithms::traits::{
    ChromosomeT, ConfigurationT, CrossoverConfig, MutationConfig, SelectionConfig, StoppingConfig,
};

fn main() {
    println!("=== Hall of Fame / Solution Archive Demo ===");
    println!();

    // Problem: maximize sum of gene values
    // 8 genes, each in range [0, 99]
    let n: i32 = 8;
    let alleles = vec![RangeGene::new(0, vec![(0_i32, 99_i32)], 0)];
    let alleles_clone = alleles.clone();

    // Hall of Fame configuration:
    // - Capacity 15 solutions
    // - No diversity filtering (Fitness with min_distance 0.0)
    let hof_config = HallOfFameConfig {
        capacity: 15,
        distance_metric: DistanceMetric::Fitness { min_distance: 0.0 },
    };

    let mut ga: Ga<RangeChromosome<i32>> = Ga::new()
        .with_genes_per_chromosome(n.try_into().unwrap())
        .with_population_size(50)
        .with_initialization_fn(move |genes_per_chromosome, _, _| {
            range_random_initialization(
                genes_per_chromosome,
                Some(&alleles_clone),
                Some(false),
            )
        })
        .with_fitness_fn(|dna: &[RangeGene<i32>]| {
            dna.iter().map(|g| *g.value() as f64).sum::<f64>()
        })
        .with_selection_method(Selection::Tournament)
        .with_crossover_method(Crossover::Uniform)
        .with_mutation_method(Mutation::Swap)
        .with_problem_solving(ProblemSolving::Maximization)
        .with_survivor_method(Survivor::Fitness)
        .with_max_generations(100)
        .with_hall_of_fame(hof_config)
        .build()
        .expect("Failed to build GA");

    println!("Running GA (100 generations, population 50)...");
    let _population = ga.run().expect("GA run failed");
    println!("Run complete.");
    println!();

    // Access the Hall of Fame
    let hof = ga.hall_of_fame().expect("Hall of Fame should be populated");

    println!("=== Hall of Fame Summary ===");
    println!("  Total archived solutions: {}", hof.len());
    println!("  Archive capacity: {}", hof.capacity());
    println!();

    // Show top 5 entries
    let top_n = 5.min(hof.len());
    println!("=== Top {} Solutions (best first) ===", top_n);
    for (rank, entry) in hof.top(top_n).iter().enumerate() {
        let dna_values: Vec<i32> = entry.chromosome.dna().iter().map(|g| *g.value()).collect();
        println!(
            "  #{:<3} | Fitness: {:>8.2} | Gen: {:<5} | DNA: {:?}",
            rank + 1,
            entry.fitness_at_addition,
            entry.generation_added,
            dna_values,
        );
    }
    println!();

    // Show best chromosome (for comparison with existing tracking)
    println!("=== Best Chromosome (existing tracking) ===");
    let best_chromosome = &_population.best_chromosome;
    let best_dna: Vec<i32> = best_chromosome.dna().iter().map(|g| *g.value()).collect();
    println!(
        "  Fitness: {:.2} | DNA: {:?}",
        best_chromosome.fitness(),
        best_dna,
    );
    println!();

    // Iterate over all entries
    println!("=== Full Archive ({}) ===", hof.len());
    let max_to_show = 10usize.min(hof.len());
    for (i, entry) in hof.iter().enumerate().take(max_to_show) {
        let dna_values: Vec<i32> = entry.chromosome.dna().iter().map(|g| *g.value()).collect();
        println!(
            "  [{:>2}] Fitness: {:>8.2} (gen {}) DNA: {:?}",
            i + 1,
            entry.fitness_at_addition,
            entry.generation_added,
            dna_values,
        );
    }
    if hof.len() > max_to_show {
        println!("  ... ({} more entries)", hof.len() - max_to_show);
    }
}
```
