```rust
let fitness_fn = |dna: &[RangeGenotype<f64>]| -> f64 {
    let a = 10.0;
    let n = dna.len() as f64;
    a * n + dna.iter().map(|g| {
        g.value.powi(2) - a * (2.0 * std::f64::consts::PI * g.value).cos()
    }).sum::<f64>()
};
```
