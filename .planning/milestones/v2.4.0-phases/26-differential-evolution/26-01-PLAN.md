```rust
struct CountingObserver {
    run_starts: AtomicUsize,
    generation_ends: AtomicUsize,
    new_bests: AtomicUsize,
    run_ends: AtomicUsize,
}

impl CountingObserver {
    fn new() -> Self {
        Self {
            run_starts: AtomicUsize::new(0),
            generation_ends: AtomicUsize::new(0),
            new_bests: AtomicUsize::new(0),
            run_ends: AtomicUsize::new(0),
        }
    }
}

impl GaObserver<RangeChromosome<f64>> for CountingObserver {
    fn on_run_start(&self) {
        self.run_starts.fetch_add(1, Ordering::Relaxed);
    }
    fn on_generation_end(&self, _stats: &GenerationStats) {
        self.generation_ends.fetch_add(1, Ordering::Relaxed);
    }
    fn on_new_best(&self, _generation: usize, _best: RangeChromosome<f64>) {
        self.new_bests.fetch_add(1, Ordering::Relaxed);
    }
    fn on_run_end(&self, _cause: TerminationCause, _all_stats: &[GenerationStats]) {
        self.run_ends.fetch_add(1, Ordering::Relaxed);
    }
}
```

```rust
#[test]
fn test_observer_hooks_fire() {
    let obs = Arc::new(CountingObserver::new());

    let config = DeConfiguration::default()
        .with_population_size(10)
        .with_max_generations(5)
        .with_problem_solving(ProblemSolving::Minimization);

    let mut engine = DeEngine::new(config, |n| random_pop(n, 5, -5.0, 5.0, 99), sphere)
        .with_observer(Arc::clone(&obs) as Arc<dyn GaObserver<RangeChromosome<f64>> + Send + Sync>);

    let _ = engine.run();

    assert_eq!(obs.run_starts.load(Ordering::Relaxed), 1, "on_run_start must fire exactly once");
    assert_eq!(obs.run_ends.load(Ordering::Relaxed), 1, "on_run_end must fire exactly once");
    assert_eq!(obs.generation_ends.load(Ordering::Relaxed), 5, "on_generation_end must fire once per generation");
    assert!(obs.new_bests.load(Ordering::Relaxed) >= 1, "on_new_best must fire at least once (initial best)");
}
```
