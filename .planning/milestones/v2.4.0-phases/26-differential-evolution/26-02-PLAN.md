```rust
fn bench_de_vs_ga(c: &mut Criterion) {
    let mut group = c.benchmark_group("de_vs_ga_sphere_5d");
    group.sample_size(10);

    // DE side: Rand1/Binomial, pop=30, 100 generations, sphere 5d
    group.bench_function("de_rand1_binomial", |b| {
        b.iter(|| {
            let config = DeConfiguration::default()
                .with_population_size(30)
                .with_max_generations(100)
                .with_mutation_strategy(DeMutationStrategy::Rand1)
                .with_problem_solving(ProblemSolving::Minimization);
            let mut engine = DeEngine::new(config, |n| make_pop(n, 5), sphere);
            engine.run()
        });
    });

    // GA side: equivalent budget (pop=30, 100 generations, sphere 5d)
    group.bench_function("ga_tournament_uniform", |b| {
        b.iter(|| {
            let mut ga = Ga::<RangeChromosome<f64>>::new();
            ga.with_population_size(30)
                .with_max_generations(100)
                .with_selection(Selection::Tournament)
                .with_crossover(Crossover::Uniform)
                .with_mutation(Mutation::Uniform)
                .with_survivor(Survivor::Fitness)
                .with_problem_solving(ProblemSolving::Minimization)
                .with_alleles(vec![(-5.0, 5.0); 5]);
            ga.run();
        });
    });

    group.finish();
}
```

Note: The Ga engine uses its built-in alleles-based initialization (no custom init_fn needed). Both engines get the same budget: 30 individuals, 100 generations, sphere(5d), minimization. The criterion output will show side-by-side wall-clock comparison.
