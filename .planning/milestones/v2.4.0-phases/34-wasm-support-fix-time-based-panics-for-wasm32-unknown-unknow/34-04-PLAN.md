```yaml
jobs:
  wasm-check:
    name: cargo check (wasm32-unknown-unknown)
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Install stable toolchain with wasm32 target
        uses: dtolnay/rust-toolchain@stable
        with:
          targets: wasm32-unknown-unknown

      - name: Cache cargo registry and target
        uses: Swatinem/rust-cache@v2
        with:
          key: wasm32-unknown-unknown

      - name: cargo check (default features)
        run: cargo check --target wasm32-unknown-unknown --lib

      - name: cargo check (serde feature)
        run: cargo check --target wasm32-unknown-unknown --lib --features serde
```

```rust
//! Smoke test for Phase 34 (WASM support).
//!
//! Verifies that a standard `Ga` run with `max_duration_secs` configured
//! terminates cleanly in a small number of generations and does not panic.
//! This exercises the non-wasm32 native path of the cfg-gated time-limit
//! check; the wasm32 path is verified by `cargo check --target
//! wasm32-unknown-unknown` in CI (`.github/workflows/wasm-check.yml`).

use genetic_algorithms::chromosomes::BinaryChromosome;
use genetic_algorithms::configuration::{LimitConfiguration, ProblemSolving, StoppingCriteria};
use genetic_algorithms::ga::Ga;
use genetic_algorithms::traits::ConfigurationT;

fn count_ones(genes: &[genetic_algorithms::genotypes::Binary]) -> f64 {
    genes.iter().filter(|g| g.id() == 1).count() as f64
}

#[test]
fn ga_runs_with_max_duration_secs() {
    // Build a tiny GA: 8 chromosomes of 8 bits, 5 generations, with a
    // generous max_duration_secs so the time-limit branch is exercised but
    // generation limit terminates first.
    let mut ga: Ga<BinaryChromosome> = Ga::new()
        .with_limit_configuration(LimitConfiguration {
            problem_solving: ProblemSolving::Maximization,
            max_generations: 5,
            population_size: 8,
            genes_per_individual: 8,
            ..Default::default()
        })
        .with_stopping_criteria(StoppingCriteria {
            max_duration_secs: Some(60.0),
            ..Default::default()
        })
        .with_fitness_fn(std::sync::Arc::new(count_ones))
        .with_alleles(vec![
            genetic_algorithms::genotypes::Binary::new(),
            genetic_algorithms::genotypes::Binary::new(),
        ])
        .build()
        .expect("valid configuration");

    let _result = ga.run();

    // Reaching this line proves no panic from Instant::now() (cfg-gated)
    // and no panic from rayon (cfg-gated). Termination cause should be
    // GenerationLimitReached because 5 generations runs faster than 60 s.
    assert_eq!(
        ga.termination_cause,
        genetic_algorithms::ga::TerminationCause::GenerationLimitReached,
        "Expected GenerationLimitReached, got {:?}",
        ga.termination_cause
    );
}
```

If `BinaryChromosome` / `Binary` import paths differ from the above (the executor must verify by reading `src/lib.rs` re-exports and `src/chromosomes/mod.rs`), adjust the imports to match the canonical re-export, but keep the test logic identical: build a 5-generation GA with `max_duration_secs: Some(60.0)`, run it, assert termination cause is `GenerationLimitReached`. Do NOT add `#[cfg(target_arch = "wasm32")]` gating to the test itself — host-target tests run only on native; the wasm32 build path is verified via `cargo check` in CI (Task 1), not via test execution.
