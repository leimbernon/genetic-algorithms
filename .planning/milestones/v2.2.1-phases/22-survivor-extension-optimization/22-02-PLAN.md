```rust
// Regrow population if extension reduced it
if self.population.chromosomes.len() < initial_population_size {
    if let Some(ref init_fn) = self.initialization_fn {
        let deficit = initial_population_size - self.population.chromosomes.len();
        let genes_per_chromosome = self.configuration.limit_configuration.genes_per_chromosome;
        let alleles_can_be_repeated = self.configuration.limit_configuration.alleles_can_be_repeated;
        let alleles_ref: Option<&[U::Gene]> = if self.alleles.is_empty() {
            None
        } else {
            Some(self.alleles.as_slice())
        };
        let ff = self.fitness_fn.as_ref().map(Arc::clone);

        let new_chromosomes: Vec<U> = (0..deficit)
            .into_par_iter()
            .map(|_| {
                let genes = init_fn(genes_per_chromosome, alleles_ref, Some(alleles_can_be_repeated));
                let mut new_chromosome = U::new();
                new_chromosome.set_dna(std::borrow::Cow::Owned(genes));
                if let Some(ref ff) = ff {
                    let ff_clone = Arc::clone(ff);
                    new_chromosome.set_fitness_fn(move |genes| ff_clone(genes));
                }
                new_chromosome.calculate_fitness();
                new_chromosome.set_age(0);
                new_chromosome
            })
            .collect();
        self.population.chromosomes.extend(new_chromosomes);
    }
}
```

Key differences from original:
1. `alleles_ref`, `genes_per_chromosome`, `alleles_can_be_repeated`, and `ff` are extracted before the closure so they can be shared across threads
2. `(0..deficit).into_par_iter().map(...).collect()` replaces the sequential for loop
3. `self.population.chromosomes.extend(new_chromosomes)` replaces individual `.push()` calls
4. The observer `on_extension_triggered` call BEFORE this block is unchanged
5. The NaN recalculation block AFTER this block is unchanged

Note: `init_fn` is `Arc<dyn Fn(...) + Send + Sync>` and `ff` is `Arc<dyn Fn(...) + Send + Sync>` so both are safe to share across rayon threads. `alleles_ref` borrows `self.alleles` which is fine since we don't mutate `self` inside the closure.
