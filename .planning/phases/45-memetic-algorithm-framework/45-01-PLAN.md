    let step = self.config.local_search_step_size;
    let dim = ind.dna().len();
    for _ in 0..self.config.local_search_steps {
        let j = rng.random_range(0..dim);
        let delta: f64 = (rng.random::<f64>() * 2.0 - 1.0) * step;
        let old_val = ind.dna()[j].de_value();
