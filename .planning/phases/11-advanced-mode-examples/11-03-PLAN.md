**Fitness function** -- greedy parallel machine scheduling:
```rust
let fitness_fn = |dna: &[RangeGenotype<i32>]| -> f64 {
    let mut machine_finish = [0u32; N_MACHINES];
    for gene in dna {
        let job = gene.value as usize;
        // Assign job to the machine with the earliest available time
        let (m, _) = machine_finish.iter().enumerate()
            .min_by_key(|(_, &t)| t).unwrap();
        machine_finish[m] += PROCESSING_TIMES[job][m];
    }
    *machine_finish.iter().max().unwrap() as f64
};
```
