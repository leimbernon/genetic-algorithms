let fitness_fn = |dna: &[RangeGenotype<f64>]| -> f64 {
    let x = dna[0].value;
    let peak = |center: f64, height: f64| -> f64 {
        height * (-((x - center).powi(2)) / (2.0 * 0.5_f64.powi(2))).exp()
    };
    peak(2.0, 1.0) + peak(5.0, 0.9) + peak(8.0, 0.8)
};
```

println!("-------------------------------------------------------");
println!("Finished. Best fitness: {:.4}", population.best_chromosome.fitness);
println!("\nTop solutions (showing population spread across peaks):");

// Collect all chromosome positions and sort
let mut positions: Vec<f64> = population.chromosomes.iter()
    .map(|c| c.dna()[0].value)
    .collect();
positions.sort_by(|a, b| a.partial_cmp(b).unwrap());

// Count solutions near each peak (within sigma_share)
let peaks = [(2.0, "Peak 1 (x=2)"), (5.0, "Peak 2 (x=5)"), (8.0, "Peak 3 (x=8)")];
for (center, label) in &peaks {
    let count = positions.iter().filter(|&&x| (x - center).abs() < SIGMA_SHARE).count();
    println!("  {}: {} individuals", label, count);
}

let peaks_found = peaks.iter()
    .filter(|(center, _)| positions.iter().any(|&x| (x - center).abs() < SIGMA_SHARE))
    .count();
if peaks_found >= 3 {
    println!("SUCCESS: Population covers all {} peaks!", peaks_found);
} else {
    println!("Found {} of 3 peaks. Try increasing population or adjusting sigma_share.", peaks_found);
}
```
