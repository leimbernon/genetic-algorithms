    Define pub(crate) fn full_tree<N: GpNode>(max_depth: usize, rng: &mut impl Rng) -> Node<N>:
      if max_depth <= 1: return Node::Terminal(N::sample_random_terminal(rng)).
      let functions = N::all_functions(); if functions is empty, return Terminal.
      pick random function f from functions.
      children = (0..f.arity()).map(|_| Box::new(full_tree(max_depth - 1, rng))).collect().
      Node::Function { value: f, children }.

    Refine grow_tree<N: GpNode>(max_depth: usize, rng: &mut impl Rng) -> Node<N>:
      if max_depth <= 1: return Node::Terminal(N::sample_random_terminal(rng)).
      let functions = N::all_functions(); if functions is empty, return Terminal.
      let p_terminal: f64 = 0.5; // default; user cannot configure per-node probability in v3.0.0
      if rng.random::<f64>() < p_terminal: return Terminal(N::sample_random_terminal(rng)).
      pick random function f. children recursive with max_depth - 1.
      Node::Function { value: f, children }.

    Define pub fn ramped_half_and_half<N: GpNode + Default>(pop_size: usize, init_max_depth: usize, rng: &mut impl Rng) -> Vec<GpChromosome<N>>:
      let depth_range = 2..=init_max_depth; // standard GP practice
      let num_depths = depth_range.clone().count().max(1);
      let per_depth = (pop_size / num_depths).max(1);
      let mut pop: Vec<GpChromosome<N>> = Vec::with_capacity(pop_size);
      for d in depth_range:
        for i in 0..per_depth:
          let root = if i % 2 == 0 { full_tree(d, rng) } else { grow_tree(d, rng) };
          pop.push(GpChromosome::with_root(Box::new(root)));
      // Fill remainder if pop_size not evenly divisible
      while pop.len() < pop_size:
        let root = grow_tree(init_max_depth, rng);
        pop.push(GpChromosome::with_root(Box::new(root)));
      pop

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| user fitness_fn → engine | User-provided closure runs inside evaluate_population; a panicking closure propagates to run() — documented |
| bloat retry → discard | Rejected offspring fall back to better parent copy; population size invariant maintained |
| GpGa::run() → selection::factory | Selection receives Vec<U>; factory only accesses fitness() — safe for GpChromosome |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-53-07 | Denial of Service | user fitness_fn panics | accept | Documented: fitness_fn must not panic; panics propagate as Rust panics (same contract as all other engines) |
| T-53-08 | Denial of Service | infinite crossover retry loop | mitigate | Hard cap of 3 retries per pair; falls back to better parent copy to keep population full |

| T-53-SC | Tampering | cargo installs | accept | No new dependencies in Wave 2 |
