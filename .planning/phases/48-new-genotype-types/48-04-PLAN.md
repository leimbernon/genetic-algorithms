pub struct MultiUniqueChromosome<T: Sync + Send + Clone + Default + Debug> {
    pub dna: Vec<UniqueGenotype<T>>,
    pub groups: Vec<Arc<[T]>>,
    pub fitness: f64,
    pub age: usize,
    #[cfg_attr(feature = "serde", serde(skip, default))]
    pub fitness_fn: FitnessFnWrapper<UniqueGenotype<T>>,
}

<!-- D-14: group_ranges() derives boundaries from group alphabet lengths -->
pub fn group_ranges(&self) -> Vec<(usize, usize)>;
// Returns [(0, g0_len-1), (g0_len, g0_len+g1_len-1), ...]

<!-- OperatorCompat: restricts to multi-group crossovers + permutation-safe defaults (D-07, A5) -->
impl<T: ...> OperatorCompat for MultiUniqueChromosome<T> {
    fn valid_crossovers() -> Option<&'static [Crossover]> {
        Some(&[Crossover::MultiGroupPmx, Crossover::MultiGroupOx,
               Crossover::Clone, Crossover::Rejuvenate])
    }
    fn valid_mutations() -> Option<&'static [Mutation]> {
        Some(&[Mutation::Insertion, Mutation::Swap, Mutation::Inversion])
    }
}

<!-- multi_group_pmx: concrete fn for MultiUniqueChromosome<T> (RESEARCH dispatch decision: concrete is simpler than introducing a new trait) -->

// Calls p1.group_ranges(); for each (start, end), calls pub(crate) pmx::pmx_build_child on the slice; assembles two children.

<!-- multi_group_ox: same shape, calls order::ox_build_child -->
pub fn multi_group_ox<T: ...>(p1: &MultiUniqueChromosome<T>, p2: &MultiUniqueChromosome<T>) -> Result<Vec<MultiUniqueChromosome<T>>, GaError>;
